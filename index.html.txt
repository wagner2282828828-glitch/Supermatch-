<script>
  const API_URL = location.protocol.startsWith("http") ? "/api/public/cantadas" : "COLOQUE_AQUI_O_ENDERECO/api/public/cantadas";

  const FRASES = ["Analisando cada detalhe...","Esquentando as ideias...","Caprichando na malícia...","Quase pronto, segura o coração..."];
  const $ = (id) => document.getElementById(id);
  let modo = "story", nivel = "safada", imagem = null, carregando = false;

  function toast(t){ const el=$("toast"); el.textContent=t; el.classList.add("show"); clearTimeout(el._t); el._t=setTimeout(()=>el.classList.remove("show"),2200); }

  function atualizar(){
    document.querySelectorAll("[data-modo]").forEach(b=>b.classList.toggle("on", b.dataset.modo===modo));
    document.querySelectorAll("[data-nivel]").forEach(b=>b.classList.toggle("on", b.dataset.nivel===nivel));
    $("dica").textContent = modo==="story" ? "Manda o story ou foto do crush" : "Manda o print da conversa";
    $("contexto").style.display = modo==="conversa" ? "block" : "none";
    $("drop").style.display = imagem ? "none" : "flex";
    $("preview").style.display = imagem ? "block" : "none";
    if (imagem) $("img").src = imagem;
    if (!carregando) $("gerar").textContent = modo==="story" ? "✨ Criar reações" : "✨ Criar respostas";
  }
  function limparResultado(){ $("res").style.display="none"; }

  function lerImagem(file){
    return new Promise((ok, erro)=>{
      const url = URL.createObjectURL(file); const img = new Image();
      img.onload = ()=>{
        const e = Math.min(1, 1600/Math.max(img.naturalWidth, img.naturalHeight));
        const c = document.createElement("canvas");
        c.width = Math.round(img.naturalWidth*e); c.height = Math.round(img.naturalHeight*e);
        c.getContext("2d").drawImage(img,0,0,c.width,c.height);
        URL.revokeObjectURL(url); ok(c.toDataURL("image/jpeg",0.85));
      };
      img.onerror = ()=>{ URL.revokeObjectURL(url); erro(new Error("Não consegui abrir essa imagem.")); };
      img.src = url;
    });
  }
  async function carregar(file){
    if (!file) return;
    if (!file.type.startsWith("image/")) return toast("Isso não é uma imagem.");
    try { imagem = await lerImagem(file); limparResultado(); atualizar(); } catch(e){ toast(e.message); }
  }

  async function gerar(){
    if (!imagem){ toast("Primeiro manda uma imagem 📸"); $("arquivo").click(); return; }
    carregando = true; let i = 0;
    const btn = $("gerar"); btn.disabled = true; $("outras").disabled = true;
    btn.innerHTML = '<span class="flick">🔥</span> ' + FRASES[0];
    const timer = setInterval(()=>{ i=(i+1)%FRASES.length; btn.innerHTML='<span class="flick">🔥</span> '+FRASES[i]; }, 2200);
    try {
      const contexto = $("contexto").value.trim();
      const r = await fetch(API_URL, { method:"POST", headers:{"Content-Type":"application/json"},
        body: JSON.stringify({ modo, nivel, imagem, ...(contexto && modo==="conversa" ? {contexto} : {}) }) });
      const d = await r.json().catch(()=>({erro:"Resposta inválida do servidor."}));
      if (!r.ok) throw new Error(d.erro || "Deu ruim. Tenta de novo.");
      mostrar(d);
    } catch(e){ toast(e.message || "Deu ruim. Tenta de novo."); }
    finally { clearInterval(timer); carregando=false; btn.disabled=false; $("outras").disabled=false; atualizar(); }
  }

  function mostrar(d){
    $("leitura").textContent = d.leitura;
    const lista = $("lista"); lista.innerHTML = "";
    d.respostas.rules?.forEach ? null : d.respostas.forEach((r, idx)=>{
      const el = document.createElement("div"); el.className="msg"; el.style.animationDelay=(idx+1)*90+"ms";
      const e = document.createElement("div"); e.className="e"; e.textContent=r.estilo;
      const p = document.createElement("p"); p.textContent=r.texto;
      const b = document.createElement("button"); b.textContent="📋"; b.setAttribute("aria-label","Copiar");
      b.onclick = async ()=>{ try{ await navigator.clipboard.writeText(r.texto); b.textContent="✓"; toast("Copiado! Agora é só mandar 😏"); setTimeout(()=>b.textContent="📋",1800);}catch{ toast("Não deu pra copiar."); } };
      el.append(e,p,b); lista.append(el);
    });
    $("res").style.display="flex";
    $("res").scrollIntoView({behavior:"smooth", block:"start"});
  }

  document.querySelectorAll("[data-modo]").forEach(b=>b.onclick=()=>{ modo=b.dataset.modo; limparResultado(); atualizar(); });
  document.querySelectorAll("[data-nivel]").forEach(b=>b.onclick=()=>{ nivel=b.dataset.nivel; atualizar(); });
  $("drop").onclick = $("trocar").onclick = ()=>$("arquivo").click();
  $("arquivo").onchange = (e)=>{ carregar(e.target.files[0]); e.target.value=""; };
  $("remover").onclick = ()=>{ imagem=null; limparResultado(); atualizar(); };
  $("drop").ondragover = (e)=>e.preventDefault();
  $("drop").ondrop = (e)=>{ e.preventDefault(); carregar(e.dataTransfer.files[0]); };
  window.addEventListener("paste", (e)=>{ const it=[...(e.clipboardData?.items||[])].find(i=>i.type.startsWith("image/")); if(it) carregar(it.getAsFile()); });
  $("gerar").onclick = gerar; $("outras").onclick = gerar;
  atualizar();
</script>
